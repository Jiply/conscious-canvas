```typescript
export const getAgentDecisions = query({

    });
```

```typescript
// Form or update an opinion
export const updateOpinion = mutation({
  handler: async (
    ctx,
    args: {
      agentId: Id<"agents">;
      topic: string;
      stance: number;
      confidence: number;
      reasoning: string;
    }
  ) => {
    const existing = await ctx.db
      .query("opinions")
      .filter((q) => q.eq(q.field("agentId"), args.agentId))
      .filter((q) => q.eq(q.field("topic"), args.topic))
      .first();

    if (existing) {
      return await ctx.db.patch(existing._id, {
        stance: args.stance,
        confidence: args.confidence,
        reasoning: args.reasoning,
        lastUpdated: Date.now(),
      });
    } else {
      return await ctx.db.insert("opinions", {
        ...args,
        lastUpdated: Date.now(),
      });
    }
  },
});

// Get all opinions for an agent
export const getAgentOpinions = query({
  handler: async (ctx, args: { agentId: Id<"agents"> }) => {
    return await ctx.db
      .query("opinions")
      .filter((q) => q.eq(q.field("agentId"), args.agentId))
      .collect();
  },
});
```

### **G. Heartbeat System (Agent Tick Processing)**

#### **File**: `convex/heartbeat.ts` (ENHANCED)

**Purpose**: Central game loop for agent updates

**Key Features**:

- **Tick-Based Processing**:
  - Regular agent updates (every N seconds)
  - Processes all agents systematically
  - Executes decisions
  - Updates agent states
  - Triggers observations and conversations

- **Agent Lifecycle per Tick**:
  1. Check current decision status
  2. Complete finished actions
  3. Make new decisions if idle
  4. Update position along path
  5. Check for nearby agents
  6. Trigger observations
  7. Evaluate conversation opportunities
  8. Update emotional state
  9. Form/update opinions

- **Heartbeat Functions**:

  ```typescript
  export const processAgentTick = mutation({
    handler: async (ctx, args: { agentId: Id<"agents"> }) => {
      const agent = await ctx.db.get(args.agentId);

      // Process current decision
      const decision = await getCurrentDecision(ctx, agent._id);
      if (decision) {
        await executeDecision(ctx, agent, decision);
      } else {
        // Make new decision if idle
        await ctx.runMutation(api.decisions.makeDecision, {
          agentId: agent._id,
        });
      }

      // Check for observations
      await checkForObservations(ctx, agent);

      // Check for conversation opportunities
      await checkForConversations(ctx, agent);

      // Update emotional state
      await updateEmotionalState(ctx, agent);
    },
  });

  // Process all agents
  export const tickAllAgents = action({
    handler: async (ctx) => {
      const agents = await ctx.runQuery(api.agents.listAgents);
      for (const agent of agents) {
        await ctx.runMutation(api.heartbeat.processAgentTick, {
          agentId: agent._id,
        });
      }
    },
  });
  ```

- **Scheduled Processing**:
  - Cron job or interval-based
  - Configurable tick rate
  - Handles large numbers of agents
  - Error resilience
